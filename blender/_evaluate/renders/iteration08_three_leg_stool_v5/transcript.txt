```
import bmesh, math
from mathutils import Vector, Matrix

def cylinder_between(name, A, B, radius, segments=24):
    axis = Vector(B) - Vector(A)
    length = axis.length
    dirn = axis.normalized()
    # rotation matrix mapping Z to dirn
    z = Vector((0,0,1))
    if abs(dirn.dot(z)) > 0.9999:
        rot = Matrix.Identity(3)
        if dirn.z < 0:
            rot = Matrix.Rotation(math.pi, 3, 'X')
    else:
        axis_rot = z.cross(dirn)
        angle = math.acos(z.dot(dirn))
        rot = Matrix.Rotation(angle, 3, axis_rot)
    # build cylinder
    mesh = bpy.data.meshes.new(name)
    bm = bmesh.new()
    # side verts
    top_ring = []
    bot_ring = []
    for i in range(segments):
        a = 2*math.pi*i/segments
        x = radius*math.cos(a); y = radius*math.sin(a)
        bot_ring.append(bm.verts.new((x,y,0)))
        top_ring.append(bm.verts.new((x,y,length)))
    for i in range(segments):
        j=(i+1)%segments
        bm.faces.new([bot_ring[i], bot_ring[j], top_ring[j], top_ring[i]])
    bm.faces.new(bot_ring)
    bm.faces.new(top_ring)
    bmesh.ops.recalc_face_normals(bm, faces=bm.faces)
    # apply rotation
    for v in bm.verts:
        v.co = rot @ v.co + Vector(A)
    bm.to_mesh(mesh)
    bm.free()
    obj = bpy.data.objects.new(name, mesh)
    link_into_scene(obj)
    return obj
```
