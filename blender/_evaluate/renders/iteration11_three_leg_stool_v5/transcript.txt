```python
def place_leg(name, theta):
    leg = add_cylinder(name, LEG_RADIUS_M, LEG_LENGTH_M, segment_count=16)
    # axis points
    p0 = Vector((FOOT_RADIUS_M*math.cos(theta), FOOT_RADIUS_M*math.sin(theta), 0.0))
    p1 = Vector((LEG_TOP_RADIUS_M*math.cos(theta), LEG_TOP_RADIUS_M*math.sin(theta), LEG_TOP_Z_M))
    d = (p1 - p0).normalized()
    # build basis
    z = d
    x = Vector((0,0,1)).cross(z)
    x.normalize()
    y = z.cross(x)
    R = Matrix((x, y, z)).transposed()  # columns x,y,z
    # base position: extend below p0
    base = p0 - d * LEG_BELOW_M
    M = Matrix.Translation(base) @ R.to_4x4()
    leg.matrix_world = M
    link_into_scene(leg)
    return leg
```
