    a = i * 2*math.pi/3
    B = Vector((FOOT_RADIUS*math.cos(a), FOOT_RADIUS*math.sin(a), 0.0))
    T = Vector((LEG_TOP_RADIUS*math.cos(a), LEG_TOP_RADIUS*math.sin(a), LEG_TOP_Z))
    d = T - B
    L = d.length
    d_hat = d.normalized()
    # rotation mapping Z to d_hat
    z = Vector((0,0,1))
    axis = z.cross(d_hat)
    if axis.length < 1e-6:
        R = Matrix.Identity(3)
        if d_hat.z < 0:
            R = Matrix.Rotation(math.pi, 3, 'X')
    else:
        angle = math.acos(d_hat.z)
        R = Matrix.Rotation(angle, 3, axis.normalized())
    M = Matrix.Translation(B) @ R.to_4x4()
    leg = add_cylinder(f"Stool_leg_{i}", LEG_RADIUS, L, SEGMENTS, (0,0,0))
    leg.matrix_world = M
    link_into_scene(leg)
