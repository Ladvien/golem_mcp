```python
def rotation_aligning_z_to(direction):
    z = mathutils.Vector((0,0,1))
    d = direction.normalized()
    # axis = z cross d, angle = angle between
    axis = z.cross(d)
    if axis.length < 1e-9:
        # d parallel to z
        return mathutils.Matrix.Identity(4) if d.z > 0 else mathutils.Matrix.Rotation(math.pi, 4, 'X')
    axis.normalize()
    angle = z.angle(d)
    return mathutils.Matrix.Rotation(angle, 4, axis)
```
